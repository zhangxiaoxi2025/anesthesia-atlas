const unit = a => { const l = Math.hypot(...a); return a.map(x => x / l); };
const sub = (a, b) => a.map((x, i) => x - b[i]);
const dot = (a, b) => a.reduce((s, x, i) => s + x * b[i], 0);
const cross = (a, b) => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const distance = (a, b) => Math.hypot(...sub(a, b));
const vertex = (g, i) => g.positions.slice(i * 3, i * 3 + 3);
const percentiles = arr => {
  const a = arr.slice().sort((x, y) => x - y);
  return Object.fromEntries([['min', 0], ['p01', .01], ['p05', .05], ['median', .5], ['p95', .95], ['p99', .99], ['max', 1]].map(([k, p]) => [k, a[Math.floor((a.length - 1) * p)]]));
};
function components(g) {
  const positions = new Map(), mapping = [];
  for (let i = 0; i < g.positions.length; i += 3) {
    const key = g.positions.slice(i, i + 3).join(',');
    if (!positions.has(key)) positions.set(key, positions.size);
    mapping.push(positions.get(key));
  }
  const parent = Array.from({ length: positions.size }, (_, i) => i);
  const find = i => { while (parent[i] !== i) { parent[i] = parent[parent[i]]; i = parent[i]; } return i; };
  for (let i = 0; i < g.indices.length; i += 3) {
    const a = mapping[g.indices[i]], b = mapping[g.indices[i + 1]], c = mapping[g.indices[i + 2]];
    parent[find(b)] = find(a); parent[find(c)] = find(a);
  }
  return new Set(parent.map((_, i) => find(i))).size;
}
function jacobian(p, id) {
  const h = 0.00005, cols = [];
  for (let d = 0; d < 3; d++) {
    const a = p.slice(), b = p.slice(); a[d] += h; b[d] -= h;
    cols.push(sub(transformPosePoint(a, 'salute', id), transformPosePoint(b, 'salute', id)).map(v => v / (2 * h)));
  }
  return dot(cols[0], cross(cols[1], cols[2]));
}
function sourceCut(g, y, returnPoints = false) {
  const intersections = [];
  for (let i = 0; i < g.indices.length; i += 3) {
    const tri = g.indices.slice(i, i + 3);
    for (const [a, b] of [[tri[0], tri[1]], [tri[1], tri[2]], [tri[2], tri[0]]]) {
      const p = vertex(g, a), q = vertex(g, b);
      if ((p[1] < y && y <= q[1]) || (q[1] < y && y <= p[1])) {
        const t = (y - p[1]) / (q[1] - p[1]); intersections.push(p.map((v, d) => v + t * (q[d] - v)));
      }
    }
  }
  if (returnPoints) return intersections;
  if (!intersections.length) return null;
  return [0, 1, 2].map(d => (Math.min(...intersections.map(p => p[d])) + Math.max(...intersections.map(p => p[d]))) / 2);
}
function crosses(g, frame) {
  let negative = false, positive = false;
  for (let i = 0; i < g.positions.length; i += 3) {
    const d = dot(sub(g.positions.slice(i, i + 3), frame.center), frame.n);
    if (d < -1e-8) negative = true;
    if (d > 1e-8) positive = true;
    if (negative && positive) return true;
  }
  return false;
}
const targetIds = ['median', 'ulnar', 'radial', 'musculocutaneous'];
const bones = [], nerves = [], jacobians = [], jacobianSamples = [];
for (let si = 0; si < source.structures.length; si++) {
  const a = source.structures[si], b = salute.structures[si], g = a.geometry, q = b.geometry;
  assert.equal(g.positions.length, q.positions.length);
  assert.deepEqual(g.indices, q.indices, 'Topology changed for ' + a.id);
  assert(q.positions.every(Number.isFinite), 'Non-finite posed geometry');
  assert.deepEqual(neutral.structures[si].geometry.positions, g.positions);
  assert.equal(hash(JSON.stringify(g)), sourceHashes.get(a.id), 'Source mutated');
  const n = g.positions.length / 3;
  for (let i = 0; i < n; i += Math.max(1, Math.floor(n / 24))) {
    const point = vertex(g, i), determinant = jacobian(point, a.kind === 'bone' ? a.id : undefined);
    jacobians.push(determinant); jacobianSamples.push({ id: a.id, point, determinant });
  }
  if (a.kind === 'bone') {
    let maxDifferenceM = 0;
    for (let i = 0; i < 100; i++) {
      const x = i * 73 % n, y = (i * 113 + 91) % n;
      maxDifferenceM = Math.max(maxDifferenceM, Math.abs(distance(vertex(g, x), vertex(g, y)) - distance(vertex(q, x), vertex(q, y))));
    }
    assert(maxDifferenceM < 1e-9, 'Rigid bone distorted: ' + a.id);
    bones.push({ id: a.id, maxSamplePairDistanceDifferenceM: maxDifferenceM });
  }
  if (targetIds.includes(a.id)) {
    const ratios = [];
    for (let i = 0; i < g.indices.length; i += 3) {
      for (const [x, y] of [[g.indices[i], g.indices[i + 1]], [g.indices[i + 1], g.indices[i + 2]], [g.indices[i + 2], g.indices[i]]]) {
        const before = distance(vertex(g, x), vertex(g, y));
        if (before > 1e-8) ratios.push(distance(vertex(q, x), vertex(q, y)) / before);
      }
    }
    const sourceComponents = components(g), posedComponents = components(q);
    assert.equal(sourceComponents, posedComponents, 'Nerve topology disconnected: ' + a.id);
    nerves.push({ id: a.id, sourceComponents, posedComponents, edgeLengthRatio: percentiles(ratios) });
  }
}

const frames = [], localDistances = [];
for (const station of [0, .25, .64, 1]) {
  const f = getProbeFrame(salute, { station });
  assert(Math.abs(Math.hypot(...f.u) - 1) < 1e-10 && Math.abs(Math.hypot(...f.v) - 1) < 1e-10 && Math.abs(Math.hypot(...f.n) - 1) < 1e-10);
  assert(Math.max(Math.abs(dot(f.u, f.v)), Math.abs(dot(f.u, f.n)), Math.abs(dot(f.v, f.n))) < 1e-10);
  assert(dot(cross(f.u, f.v), f.n) > .999999999);
  frames.push({ station, sourceY: f.sourceY, sourceVesselId: f.sourceVesselId,
    center: f.center, u: f.u, v: f.v, n: f.n, surfaceStatus: f.surfaceStatus,
    intersectingTargetMeshes: targetIds.filter(id => crosses(salute.structures.find(s => s.id === id).geometry, f)) });
  const artery = sourceCut(source.structures.find(s => s.id === f.sourceVesselId).geometry, f.sourceY);
  for (const id of targetIds) {
    const center = sourceCut(source.structures.find(s => s.id === id).geometry, f.sourceY);
    if (!center) { localDistances.push({ station, id, status: 'source-nerve-not-at-this-level' }); continue; }
    const before = distance(center, artery), after = distance(transformPosePoint(center), transformPosePoint(artery));
    localDistances.push({ station, id, sourceCenterDistanceMm: before * 1000,
      posedCenterDistanceMm: after * 1000, ratio: after / before });
  }
}
